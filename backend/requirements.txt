# CA bundle for outbound HTTPS (OSRM, Nominatim, official ferry sources).
# Hosts whose OpenSSL ships no trust store — python.org macOS builds, slim
# containers — otherwise fail every request and silently degrade to estimates.
certifi==2026.7.22
# IANA time-zone database for zoneinfo. Kept unconditional rather than marked
# Windows-only: services/multimodal_router.py calls ZoneInfo("Asia/Jakarta") at
# import time, so a host without a system tz database fails to start at all.
# Windows never ships one, and slim Linux images frequently omit it — the same
# trap as the CA bundle above. It is a small pure-Python wheel; zoneinfo prefers
# the system database when present, so installing it everywhere costs nothing.
tzdata==2026.3
